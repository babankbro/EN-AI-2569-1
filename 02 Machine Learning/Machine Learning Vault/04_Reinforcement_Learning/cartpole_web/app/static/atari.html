def create_q_model():   # Keras
    return keras.Sequential([
        layers.Lambda(lambda t: keras.ops.transpose(t, [0, 2, 3, 1]),
                      output_shape=(84, 84, 4), input_shape=(4, 84, 84)),
        layers.Conv2D(32, 8, strides=4, activation="relu"),
        layers.Conv2D(64, 4, strides=2, activation="relu"),
        layers.Conv2D(64, 3, strides=1, activation="relu"),
        layers.Flatten(),
        layers.Dense(512, activation="relu"),
        layers.Dense(num_actions, activation="linear"),
    ])

feature maps แต่ละช่องถูก normalize แยกกัน (สว่าง = activation สูง) · conv ชั้นแรกจับขอบ/ตำแหน่งของ paddle ลูกบอล และอิฐ ชั้นลึกขึ้นรวมเป็นรูปแบบที่ซับซ้อนขึ้น · ตอน network ยังไม่ได้ฝึก feature maps ยังเป็นแค่ผลจากน้ำหนักสุ่ม

⑦ ต้องฝึกนานแค่ไหน?

ตัวอย่าง Keras ระบุว่าได้ผลดีที่ประมาณ 10 ล้าน frames ใช้เวลาไม่ถึง 24 ชั่วโมงบนเครื่องสมัยใหม่ (paper ของ DeepMind ฝึก 50 ล้าน frames) · ในหน้านี้ใช้ CPU ได้ราว ~120 frames/s → 10 ล้าน frames ≈ 23 ชั่วโมง

การกด Train ในหน้าเว็บไม่กี่นาทีจึงเห็นแค่ กลไก ของ DQN (ช่วง 50,000 frames แรกยังเป็นการสุ่มล้วน) ถ้าต้องการ agent ที่เล่นเก่ง ให้ฝึกแบบ offline ใน container แล้วกด 📂 Load checkpoint:

docker compose exec cartpole python -m app.atari_train --frames 2000000
# ฝึกต่อจาก checkpoint เดิม
docker compose exec cartpole python -m app.atari_train --frames 2000000 --resume

replay memory ในหน้าเว็บลดเหลือ 10,000 (Keras ใช้ 100,000 ≈ 5.6 GB RAM) · script offline ใช้ 50,000 เป็นค่าเริ่มต้น (--memory)
